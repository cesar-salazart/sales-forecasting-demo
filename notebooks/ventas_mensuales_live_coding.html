Pruebas de estacionariedad: ADF y KPSS¶

ADF tiene como hipótesis nula una raíz unitaria; KPSS, estacionariedad alrededor del componente determinista especificado. En la serie original se prueban constante y tendencia lineal; la primera diferencia se prueba alrededor de una constante. Esto orienta d sin diferenciar de más. La ACF/PACF de una serie estacionaria ayudan a proponer q y p, respectivamente; no son pruebas de hipótesis.

In [70]:
from statsmodels.tsa.stattools import kpss
from statsmodels.graphics.tsaplots import plot_acf

serie = agg_sales.dropna().astype(float)

def probar_estacionariedad(serie, nombre, regression="c"):
    resultado_adf = adfuller(serie, regression=regression, autolag="AIC")
    resultado_kpss = kpss(serie, regression=regression, nlags="auto")
    print(f"{nombre} (n={len(serie)})")
    print(f"ADF: estadístico={resultado_adf[0]:.3f}, p-valor={resultado_adf[1]:.4g}; H0: raíz unitaria")
    p_kpss = ">=0.10" if resultado_kpss[1] >= 0.10 else f"{resultado_kpss[1]:.4g}"
    referencia = "una tendencia lineal" if regression == "ct" else "una constante"
    print(f"KPSS: estadístico={resultado_kpss[0]:.3f}, p-valor={p_kpss}; H0: estacionaria alrededor de {referencia}")
    print()

probar_estacionariedad(serie, "Ventas semanales (constante)")
probar_estacionariedad(serie, "Ventas semanales (tendencia)", regression="ct")
serie_diff = serie.diff().dropna()
probar_estacionariedad(serie_diff, "Primera diferencia")

fig, axes = plt.subplots(1, 2, figsize=(14, 4))
plot_acf(serie_diff, lags=60, ax=axes[0])
plot_pacf(serie_diff, lags=60, method="ywm", ax=axes[1])
axes[0].set_title("ACF de la primera diferencia (incluye lag 52)")
axes[1].set_title("PACF de la primera diferencia (incluye lag 52)")
plt.tight_layout()
plt.show()

# SARIMA=(p,d,q)x(P,D,Q,S); para estacionalidad anual en datos semanales, S=52.
# Ejemplo de estructura: SARIMA=(p,1,q)x(0,1,0,52), solo si se justifica D=1.
import warnings
# Suprime las warnings
warnings.filterwarnings("ignore", category=UserWarning, module="statsmodels")
warnings.filterwarnings("ignore", category=FutureWarning)

# Completamos el código en vivo
mejor_aic = float('inf')
mejores_parametros = None

for p, q in itertools.product(range(3), range(4)):
    try:
        modelo = ARIMA(agg_sales_train, order=(p, 1, q), seasonal_order=(0, 1, 0, 52))
        resultado = modelo.fit()
        if resultado.aic < mejor_aic:
            mejor_aic = resultado.aic
            mejores_parametros = (p, q)
    except Exception as e:
        continue
print(f"Mejores parámetros: {mejores_parametros}")
print(f"Mejor AIC: {mejor_aic:.2f}")
